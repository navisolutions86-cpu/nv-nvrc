Tài khoản của bạn chỉ có quyền xem trạng thái kết nối. Liên hệ Admin Tổng hoặc Lãnh đạo có quyền "Cấu hình Supabase" để chỉnh sửa thông số kết nối.

Sử dụng khóa anon / public (không dùng service_role key)

Kết nối thất bại (HTTP 404): {"code":"PGRST205","details":null,"hint":null,"message":"Could not find the table 'public.nv_nvrc_app_state' in the schema cache"}
SQL Setup (Dán vào Supabase → SQL Editor và bấm Run 1 lần duy nhất) 
-- 1. Tạo bảng lưu trữ trạng thái đồng bộ Real-time (NAVI - NVRC)
CREATE TABLE IF NOT EXISTS public.nv_nvrc_app_state (
    id TEXT PRIMARY KEY,
    payload JSONB NOT NULL,
    updated_by TEXT,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 2. Bật Row Level Security (RLS) & chính sách cho Anon / Authenticated
ALTER TABLE public.nv_nvrc_app_state ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Allow read app state" ON public.nv_nvrc_app_state;
CREATE POLICY "Allow read app state" ON public.nv_nvrc_app_state FOR SELECT TO anon, authenticated USING (true);

DROP POLICY IF EXISTS "Allow insert app state" ON public.nv_nvrc_app_state;
CREATE POLICY "Allow insert app state" ON public.nv_nvrc_app_state FOR INSERT TO anon, authenticated WITH CHECK (true);

DROP POLICY IF EXISTS "Allow update app state" ON public.nv_nvrc_app_state;
CREATE POLICY "Allow update app state" ON public.nv_nvrc_app_state FOR UPDATE TO anon, authenticated USING (true) WITH CHECK (true);

-- 3. Kích hoạt Real-time đồng bộ tức thì giữa các trình duyệt
DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_publication_tables
        WHERE pubname = 'supabase_realtime' AND schemaname = 'public' AND tablename = 'nv_nvrc_app_state'
    ) THEN
        ALTER PUBLICATION supabase_realtime ADD TABLE public.nv_nvrc_app_state;
    END IF;
END $$;
